Cohere Integration

Use this integration when your app is implemented in Python or TypeScript.

Setup

  1. Install SDK
pip install progress-observability
  1. Instrument your app
Observability.instrument(
 app_name=os.getenv("OBSERVABILITY_APP_NAME"),
 api_key=os.getenv("OBSERVABILITY_API_KEY")
)
  1. Complete example
import json
import os
from typing import Any

import cohere
from dotenv import load_dotenv

from progress.observability import Observability

load_dotenv(override=True)

# LLM INSTRUMENTATION
Observability.instrument(
   app_name=os.getenv("OBSERVABILITY_APP_NAME", "weather-agent-cohere"),
   api_key=os.getenv("OBSERVABILITY_API_KEY"),
)


def get_weather(location: str) -> str:
   """Get the weather for a given location."""
   return f"The weather in {location} is cloudy with a high of 15°C."


def extract_text_content(content: Any) -> str:
   if isinstance(content, str):
       return content

   if not content:
       return ""

   parts: list[str] = []
   for item in content:
       text = getattr(item, "text", None)
       if isinstance(text, str):
           parts.append(text)
           continue

       if isinstance(item, dict) and isinstance(item.get("text"), str):
           parts.append(item["text"])

   return "\n".join(parts)


def main() -> None:
   api_key = os.getenv("COHERE_API_KEY")
   model = os.getenv("COHERE_MODEL", "command-a-03-2025")

   if not api_key:
       raise RuntimeError("Missing Cohere config. Set COHERE_API_KEY.")

   client = cohere.ClientV2(api_key=api_key)

   messages = [{"role": "user", "content": "What's the weather in Paris?"}]

   while True:
       response = client.chat(
           model=model,
           temperature=0.7,
           max_tokens=256,
           messages=[
               {"role": "system", "content": "You are a helpful AI assistant."},
               *messages,
           ],
           tools=[
               {
                   "type": "function",
                   "function": {
                       "name": "get_weather",
                       "description": "Get the weather for a given location.",
                       "parameters": {
                           "type": "object",
                           "properties": {"location": {"type": "string"}},
                           "required": ["location"],
                       },
                   },
               }
           ],
           tool_choice="auto",
       )

       message = getattr(response, "message", None)
       tool_calls = getattr(message, "tool_calls", None) or []

       if not tool_calls:
           print(extract_text_content(getattr(message, "content", None)))
           break

       messages.append(
           {
               "role": "assistant",
               "content": extract_text_content(getattr(message, "content", None)),
               "tool_calls": [
                   tool_call.model_dump()
                   if hasattr(tool_call, "model_dump")
                   else tool_call
                   for tool_call in tool_calls
               ],
           }
       )

       for tool_call in tool_calls:
           function_data = getattr(tool_call, "function", None)
           arguments_raw = getattr(function_data, "arguments", None) if function_data else None
           args: dict[str, Any] = {}

           if isinstance(arguments_raw, str) and arguments_raw:
               args = json.loads(arguments_raw)
           elif isinstance(arguments_raw, dict):
               args = arguments_raw

           location = args.get("location", "")

           messages.append(
               {
                   "role": "tool",
                   "tool_call_id": getattr(tool_call, "id", ""),
                   "content": get_weather(location),
               }
           )


try:
   main()
finally:
   Observability.shutdown()

  1. Complete example
import '@progress/observability/register/hooks';
import 'dotenv/config';

import { CohereClientV2 } from 'cohere-ai';
import { Observability } from '@progress/observability';

async function initializeObservability() {
  await Observability.instrument({
    appName: process.env.OBSERVABILITY_APP_NAME ?? 'weather-agent-cohere-sdk',
    apiKey: process.env.OBSERVABILITY_API_KEY,
  });
}

const cohereModel = process.env.COHERE_MODEL || 'command-a-03-2025';

async function main() {
  await initializeObservability();

  try {
    const client = new CohereClientV2({
      token: process.env.COHERE_API_KEY,
    });

    const result = await client.chat({
      model: cohereModel,
      messages: [
        { role: 'system', content: 'You are a helpful AI assistant.' },
        { role: 'user', content: "What's the weather in Paris?" },
      ],
      temperature: 0.7,
    });

    const textResponse =
      result.message?.content
        ?.map((item) => {
          if ('text' in item && typeof item.text === 'string') {
            return item.text;
          }
          return '';
        })
        .filter(Boolean)
        .join('\n') ?? '';

    console.log(textResponse);
  } finally {
    await Observability.shutdown();
  }
}

main().catch((error) => {
  console.error(error);
  process.exitCode = 1;
});
