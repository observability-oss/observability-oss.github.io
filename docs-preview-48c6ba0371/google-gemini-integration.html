Google Gemini Integration

Use this integration when your app is implemented in Python or TypeScript.

Setup

  1. Install SDK
pip install progress-observability
  1. Instrument your app
Observability.instrument(
 app_name=os.getenv("OBSERVABILITY_APP_NAME"),
 api_key=os.getenv("OBSERVABILITY_API_KEY")
)
  1. Complete example
import json
import os
from typing import Any

from google import genai
from google.genai import types
from dotenv import load_dotenv

from progress.observability import Observability

load_dotenv(override=True)

# LLM INSTRUMENTATION
Observability.instrument(
    app_name=os.getenv("OBSERVABILITY_APP_NAME", "weather-agent-gemini"),
    api_key=os.getenv("OBSERVABILITY_API_KEY"),
)


def get_weather(location: str) -> str:
   """Get the weather for a given location."""
   return f"The weather in {location} is cloudy with a high of 15°C."


def extract_text_content(response: Any) -> str:
   candidates = getattr(response, "candidates", None) or []
   parts: list[str] = []
   for candidate in candidates:
       content = getattr(candidate, "content", None)
       content_parts = getattr(content, "parts", None) or []
       for part in content_parts:
           part_text = getattr(part, "text", None)
           if isinstance(part_text, str) and part_text.strip():
               parts.append(part_text)

   if parts:
       return "\n".join(parts)

   return "No text response returned by Gemini."


def main() -> None:
   api_key = os.getenv("GOOGLE_API_KEY")
   model = os.getenv("GEMINI_MODEL", "gemini-2.5-flash")

   if not api_key:
       raise RuntimeError("Missing Gemini config. Set GOOGLE_API_KEY.")

   client = genai.Client(api_key=api_key)

   messages = [{"role": "user", "content": "What's the weather in Paris?"}]

   while True:
       gemini_contents: list[types.Content] = []
       for message in messages:
           role = message.get("role", "user")

           if role == "user":
               gemini_contents.append(
                   types.Content(
                       role="user",
                       parts=[types.Part(text=str(message.get("content", "")))],
                   )
               )
           elif role == "assistant":
               parts: list[types.Part] = []
               tool_calls = message.get("tool_calls", [])
               for tool_call in tool_calls:
                   function_name = tool_call.get("function", {}).get("name", "")
                   function_args = tool_call.get("function", {}).get("arguments", {})

                   if isinstance(function_args, str) and function_args:
                       try:
                           function_args = json.loads(function_args)
                       except json.JSONDecodeError:
                           function_args = {}

                   parts.append(
                       types.Part.from_function_call(
                           name=function_name,
                           args=function_args if isinstance(function_args, dict) else {},
                       )
                   )

               assistant_text = str(message.get("content", ""))
               if assistant_text:
                   parts.insert(0, types.Part(text=assistant_text))

               if parts:
                   gemini_contents.append(types.Content(role="model", parts=parts))
           elif role == "tool":
               tool_name = str(message.get("tool_name", "get_weather"))
               tool_result = str(message.get("content", ""))
               gemini_contents.append(
                   types.Content(
                       role="tool",
                       parts=[
                           types.Part.from_function_response(
                               name=tool_name,
                               response={"result": tool_result},
                           )
                       ],
                   )
               )

       response = client.models.generate_content(
           model=model,
           contents=gemini_contents,
           config=types.GenerateContentConfig(
               system_instruction="You are a helpful AI assistant.",
               temperature=0.7,
               max_output_tokens=256,
               tools=[
                   types.Tool(
                       function_declarations=[
                           types.FunctionDeclaration(
                               name="get_weather",
                               description="Get the weather for a given location.",
                               parameters={
                                   "type": "object",
                                   "properties": {
                                       "location": {"type": "string"},
                                   },
                                   "required": ["location"],
                               },
                           )
                       ]
                   )
               ],
           ),
       )

       function_calls = getattr(response, "function_calls", None) or []

       if not function_calls:
           print(extract_text_content(response))
           break

       messages.append(
           {
               "role": "assistant",
               "content": extract_text_content(response),
               "tool_calls": [
                   {
                       "id": getattr(function_call, "id", ""),
                       "type": "function",
                       "function": {
                           "name": getattr(function_call, "name", ""),
                           "arguments": getattr(function_call, "args", {}),
                       },
                   }
                   for function_call in function_calls
               ],
           }
       )

       for function_call in function_calls:
           arguments_raw: Any = getattr(function_call, "args", None)
           args: dict[str, Any] = {}

           if isinstance(arguments_raw, str) and arguments_raw:
               args = json.loads(arguments_raw)
           elif isinstance(arguments_raw, dict):
               args = arguments_raw

           location = args.get("location", "")

           messages.append(
               {
                   "role": "tool",
                   "tool_call_id": getattr(function_call, "id", ""),
                   "tool_name": getattr(function_call, "name", "get_weather"),
                   "content": get_weather(location),
               }
           )


try:
    main()
finally:
    Observability.shutdown()

  1. Complete example
import '@progress/observability/register/hooks';
import 'dotenv/config';

import { GoogleGenerativeAI } from '@google/generative-ai';
import { Observability } from '@progress/observability';

async function initializeObservability() {
  await Observability.instrument({
    appName: process.env.OBSERVABILITY_APP_NAME ?? 'weather-agent-gemini-sdk',
    apiKey: process.env.OBSERVABILITY_API_KEY,
  });
}

const geminiModel = process.env.GEMINI_MODEL || 'gemini-2.5-flash';

async function main() {
  await initializeObservability();

  try {
    const client = new GoogleGenerativeAI(process.env.GOOGLE_API_KEY || '');
    const model = client.getGenerativeModel({ model: geminiModel });

    const result = await model.generateContent(
      "You are a helpful AI assistant. What's the weather in Paris?"
    );

    console.log(result.response.text());
  } finally {
    await Observability.shutdown();
  }
}

main().catch((error) => {
  console.error(error);
  process.exitCode = 1;
});
